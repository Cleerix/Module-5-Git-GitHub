# create in working directory in git bash a new file with touch or
# with content with echo "text" > <file_name> 
# to add content in next line with >>

# git add <file_name> to add it to the staging area

# git status to see the actual status of the files

# git commit (if u want to add comments manually in the default editor)
# git commit -m "comment" short cut

# git push to get it to gitHub

# git ls-files shows all files, which are saved in the staging area

# then u need git commit -m "comment" again
